# NextDNS Panel – Tarayıcıda çalışan AI

Bu sürüm yalnızca GitHub Pages ile çalışacak şekilde tasarlanmıştır. Cloudflare Worker, sunucu veya API anahtarı gerekmez.

## Nasıl çalışır?

- Sohbet asistanı, Hugging Face Transformers.js kitaplığını CDN üzerinden yükler.
- Qwen2.5-0.5B-Instruct modeli kullanıcının tarayıcısında çalışır.
- Model ilk kullanımda internet üzerinden indirilir ve tarayıcı önbelleğinde saklanabilir.
- WebGPU destekleniyorsa GPU kullanılır; başarısız olursa WASM ile denenir.
- NextDNS API anahtarı sohbet modeline gönderilmez.

## Bilinen sınırlamalar

- Model dosyaları büyük olabilir; ilk açılış ve ilk yanıt zaman alabilir.
- Performans cihazın RAM, işlemci/GPU ve tarayıcı desteğine bağlıdır.
- Bu küçük yerel model, bulut tabanlı büyük modeller kadar güçlü olmayabilir.
- GitHub Pages statik barındırma sunduğundan, API anahtarı gizleyerek OpenAI gibi bulut modellerini doğrudan çağırmak mümkün değildir. Bu nedenle bu sürüm bulut OpenAI API'sini kullanmaz.

  async function loadModel() {
    if (!generatorPromise) {
      generatorPromise = (async () => {
        setStatus(getLang().loading);

        const lib = await import(LIBRARY_URL);

        lib.env.allowLocalModels = false;
        lib.env.useBrowserCache = true;

        // 2–4 GB RAM'li cihazlarda WASM'i tek iş parçacığıyla çalıştır.
        // Daha güçlü cihazlarda WebGPU tercih edilebilir.
        const lowMemory = Number(navigator.deviceMemory || 0) > 0 && Number(navigator.deviceMemory) <= 4;
        try {
          lib.env.backends.onnx.wasm.numThreads = 1;
          // Eski işlemciler/tarayıcılar SIMD desteklemeyebilir.
          // SIMD kapalı WASM ikilisini zorla; aksi halde backend açılamaz.
          lib.env.backends.onnx.wasm.simd = false;
        } catch (configError) {
          console.warn('WASM uyumluluk ayarı uygulanamadı:', configError);
        }

        const onProgress = (info) => {
          if (!info) return;

          if (
            info.status === 'progress' &&
            Number.isFinite(info.progress)
          ) {
            setStatus(
              getLang().downloading +
              ' ' +
              Math.round(info.progress) +
              '%'
            );
          } else if (
            info.status === 'initiate' ||
            info.status === 'download'
          ) {
            setStatus(getLang().downloading);
          } else if (info.status === 'ready') {
            setStatus(getLang().ready);
          }
        };

        const preferred = navigator.gpu && !lowMemory
          ? 'webgpu'
          : 'wasm';

        try {
          const pipe = await lib.pipeline(
            'text-generation',
            MODEL,
            {
              device: preferred,
              dtype: 'q4',
              progress_callback: onProgress
            }
          );

          setStatus(getLang().ready);
          return pipe;

        } catch (firstError) {
          if (preferred !== 'webgpu') {
            throw firstError;
          }

          setStatus(getLang().fallback);

          const pipe = await lib.pipeline(
            'text-generation',
            MODEL,
            {
              device: 'wasm',
              dtype: 'q4',
              progress_callback: onProgress
            }
          );

          setStatus(getLang().ready);
          return pipe;
        }

      })().catch((error) => {
        generatorPromise = null;
        throw error;
      });
    }

    return generatorPromise;

    function renderFaq(filter = '') {
      const query = filter.trim().toLocaleLowerCase('tr');
      const matches = faqQuestions.filter(entry =>
        entry.q.toLocaleLowerCase('tr').includes(query)
      );
      faqPanel.replaceChildren();

      if (!matches.length) {
        const empty = document.createElement('div');
        empty.textContent = 'Eşleşen soru bulunamadı.';
        empty.style.cssText = 'padding:12px;color:#a1a1aa;font-size:12px';
        faqPanel.appendChild(empty);
        return;
      }

      matches.forEach((entry) => {
        const item = document.createElement('button');
        item.type = 'button';
        item.setAttribute('role', 'option');
        item.textContent = entry.q;
        item.style.cssText = 'display:block;width:100%;padding:10px 12px;text-align:left;background:transparent;color:#e4e4e7;font-size:12px;line-height:1.4;border:0;border-bottom:1px solid #27272a;cursor:pointer';
        item.addEventListener('mouseenter', () => item.style.background = '#27272a');
        item.addEventListener('mouseleave', () => item.style.background = 'transparent');
        item.addEventListener('click', () => {
          faqPanel.hidden = true;
          input.value = '';
          bubble(entry.q, 'user');
          bubble(entry.a, 'assistant');
          setStatus(getLang().ready);
        });
        faqPanel.appendChild(item);
      });
    }

            : getLang().error + ' (' + errorMessage + ')';

          history.pop();

        } finally {
          send.disabled = false;
          input.disabled = false;
          input.focus();
        }
      });

    paint();

    document
      .querySelectorAll(
        '#lang-select,#lang-select-login'

        element.addEventListener('change', paint);
      });
  }

  if (document.readyState === 'loading') {

  } else {

const corsHeaders = (origin, allowed) => ({
  "Access-Control-Allow-Origin": allowed,
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type",
  "Vary": "Origin"
});

export default {
  async fetch(request, env) {
    const allowedOrigin = env.ALLOWED_ORIGIN || "https://bthn1.github.io";
    const origin = request.headers.get("Origin") || "";
    const cors = corsHeaders(origin, allowedOrigin);

    if (request.method === "OPTIONS") {
      return new Response(null, { status: 204, headers: cors });
    }

    if (origin && origin !== allowedOrigin) {
      return Response.json({ error: "Origin not allowed" }, { status: 403, headers: cors });
    }

    const url = new URL(request.url);
    if (url.pathname !== "/chat" || request.method !== "POST") {
      return Response.json({ error: "Not found. Send POST requests to /chat." }, { status: 404, headers: cors });
    }

    if (!env.OPENAI_API_KEY) {
      return Response.json({ error: "AI service is not configured. Add OPENAI_API_KEY as a Worker Secret." }, { status: 503, headers: cors });
    }

    try {
      const body = await request.json();
      const messages = Array.isArray(body.messages) ? body.messages.slice(-12) : [];
      if (
        !messages.length ||
        messages.some(m =>
          !m || !["user", "assistant"].includes(m.role) ||
          typeof m.content !== "string" || m.content.length > 2000
        )
      ) {
        return Response.json({ error: "Invalid messages" }, { status: 400, headers: cors });
      }

      const language = String(body.language || "tr").slice(0, 8);
      const instructions =
        "You are NextDNS Panel's helpful technical assistant. Answer in the user's selected language (" +
        language +
        "). Explain NextDNS, DNS, privacy, dashboard navigation and troubleshooting clearly. " +
        "Never claim you can see or change the user's live NextDNS account. Never ask for API keys, " +
        "passwords, WireGuard private keys, or other secrets. Do not request or expose personal DNS logs. " +
        "For changes to DNS settings, explain the steps and ask the user to perform them in the panel. " +
        "Be concise and age-appropriate.";

      const upstream = await fetch("https://api.openai.com/v1/responses", {
        method: "POST",
        headers: {
          "Authorization": "Bearer " + env.OPENAI_API_KEY,
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          model: env.OPENAI_MODEL || "gpt-4.1-mini",
          instructions,
          input: messages.map(m => ({
            role: m.role,
            content: [{ type: "input_text", text: m.content }]
          })),
          max_output_tokens: 700
        })
      });

      const result = await upstream.json();
      if (!upstream.ok) {
        return Response.json(
          { error: "OpenAI request failed", detail: result.error?.message || "Upstream error" },
          { status: 502, headers: cors }
        );
      }

      const reply = (result.output || [])
        .flatMap(item => item.content || [])
        .filter(part => part.type === "output_text")
        .map(part => part.text)
        .join("\n");

      return Response.json(
        { reply },
        { headers: { ...cors, "Cache-Control": "no-store" } }
      );
    } catch {
      return Response.json({ error: "Invalid request" }, { status: 400, headers: cors });
    }
  }
};
