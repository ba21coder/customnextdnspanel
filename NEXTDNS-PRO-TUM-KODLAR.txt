name: Build NextDNS Mobile APK

on:
  push:
    branches: [main]
    paths:
      - "android-mobile/**"
      - ".github/workflows/android-mobile-apk.yml"
      - "index.html"
      - "app.js"
      - "style.css"
      - "pro-features.js"
      - "wireguard.js"
      - "ai-assistant.js"
      - "ai-worker.js"
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: android-mobile
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      - name: Set up Java
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"
      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: "8.9"
      - name: Build mobile APK
        run: gradle assembleDebug
      - name: Rename APK
        run: cp app/build/outputs/apk/debug/app-debug.apk app/build/outputs/apk/debug/NextDNS-PRO-Mobile.apk
      - name: Upload APK artifact
        uses: actions/upload-artifact@v4
        with:
          name: nextdns-mobile-apk
          path: android-mobile/app/build/outputs/apk/debug/NextDNS-PRO-Mobile.apk
      - name: Publish latest mobile APK
        uses: softprops/action-gh-release@v2
        with:
          tag_name: mobile-latest
          name: NextDNS PRO Mobile - Latest
          files: android-mobile/app/build/outputs/apk/debug/NextDNS-PRO-Mobile.apk
          overwrite_files: true

name: Chrome Extension
on:
  push:
    paths:
      - "chrome-extension/**"
      - ".github/workflows/chrome-extension.yml"
  workflow_dispatch:
permissions:
  contents: write
jobs:
  package:
    runs-on: ubuntu-latest
    steps:
      - name: Depoyu indir
        uses: actions/checkout@v4
      - name: Eklenti ZIP dosyasını hazırla
        run: |
          cd chrome-extension
          zip -r ../NextDNS-PRO-Chrome.zip .
      - name: GitHub Releases dosyasını yayımla
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          TAG="chrome-extension-latest"
          if gh release view "$TAG" >/dev/null 2>&1; then
            gh release upload "$TAG" NextDNS-PRO-Chrome.zip --clobber
          else
            gh release create "$TAG" NextDNS-PRO-Chrome.zip --title "NextDNS PRO Chrome Eklentisi" --notes "NextDNS PRO Chrome eklentisi. ZIP dosyasını çıkarıp chrome://extensions üzerinden Paketlenmemiş öğe yükle ile klasörü seçin."
          fi

name: Build NextDNS PRO Linux App

on:
  push:
    branches: [main]
    paths:
      - "desktop-app/**"
      - ".github/workflows/linux-desktop.yml"
      - "index.html"
      - "app.js"
      - "style.css"
      - "pro-features.js"
      - "wireguard.js"
      - "ai-assistant.js"
      - "ai-worker.js"
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: "20"
      - name: Install system dependencies
        run: |
          sudo apt-get update
          sudo apt-get install -y libfuse2
      - name: Install dependencies
        working-directory: desktop-app
        run: npm install
      - name: Build Linux packages
        working-directory: desktop-app
        run: npm run dist:linux
      - name: Upload Linux installers
        uses: actions/upload-artifact@v4
        with:
          name: nextdns-pro-linux-x64
          path: |
            desktop-app/dist/NextDNS-PRO-Linux-x86_64.AppImage
            desktop-app/dist/NextDNS-PRO-Linux-amd64.deb
          if-no-files-found: error
      - name: Publish Linux installers
        uses: softprops/action-gh-release@v2
        with:
          tag_name: linux-latest
          name: NextDNS PRO Linux - Latest
          make_latest: false
          files: |
            desktop-app/dist/NextDNS-PRO-Linux-x86_64.AppImage
            desktop-app/dist/NextDNS-PRO-Linux-amd64.deb
          overwrite_files: true

# NextDNS Panel – Tarayıcıda çalışan AI

Bu sürüm yalnızca GitHub Pages ile çalışacak şekilde tasarlanmıştır. Cloudflare Worker, sunucu veya API anahtarı gerekmez.

## Nasıl çalışır?

- Sohbet asistanı, Hugging Face Transformers.js kitaplığını CDN üzerinden yükler.
- Qwen2.5-0.5B-Instruct modeli kullanıcının tarayıcısında çalışır.
- Model ilk kullanımda internet üzerinden indirilir ve tarayıcı önbelleğinde saklanabilir.
- WebGPU destekleniyorsa GPU kullanılır; başarısız olursa WASM ile denenir.
- NextDNS API anahtarı sohbet modeline gönderilmez.

## Bilinen sınırlamalar

- Model dosyaları büyük olabilir; ilk açılış ve ilk yanıt zaman alabilir.
- Performans cihazın RAM, işlemci/GPU ve tarayıcı desteğine bağlıdır.
- Bu küçük yerel model, bulut tabanlı büyük modeller kadar güçlü olmayabilir.
- GitHub Pages statik barındırma sunduğundan, API anahtarı gizleyerek OpenAI gibi bulut modellerini doğrudan çağırmak mümkün değildir. Bu nedenle bu sürüm bulut OpenAI API'sini kullanmaz.
